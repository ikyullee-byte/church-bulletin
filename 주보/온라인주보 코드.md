        let allQuizzes = [];

        function initQuiz() {
            const lines = rawQuizText.split('\n');
            for(let line of lines) {
                if(line.includes('/ 정답:')) {
                    let parts = line.split('/ 정답:');
                    let qText = parts[0].trim().replace(/^\d+\.\s*/, '');
                    let aText = parts[1].trim();
                    allQuizzes.push({ q: qText, a: aText });
                }
            }
            refreshQuiz();
        }

        function refreshQuiz() {
            const container = document.getElementById('quiz-container');
            container.innerHTML = '';
            
            let shuffled = [...allQuizzes].sort(() => 0.5 - Math.random());
            let selected = shuffled.slice(0, 10);

            selected.forEach((quiz, idx) => {
                let html = `
                <div class="quiz-item" onclick="toggleQuiz(this)">
                    <div class="quiz-question"><span class="q-mark">Q${idx+1}.</span> <div>${quiz.q}</div></div>
                    <div class="quiz-answer">정답: ${quiz.a}</div>
                </div>`;
                container.innerHTML += html;
            });
        }

        function initScratchCard() {
            const canvas = document.getElementById('scratch-canvas');
            if(!canvas) return;
            const ctx = canvas.getContext('2d');
            const verses = ["항상 기뻐하라 (살전 5:16)", "여호와는 나의 목자시니 (시 23:1)", "내가 너와 함께 함이라 (사 41:10)"];
            document.getElementById('manna-ref').innerText = verses[Math.floor(Math.random() * verses.length)];
            
            canvas.width = canvas.parentElement.offsetWidth;
            canvas.height = canvas.parentElement.offsetHeight;
            
            // 은색 코팅 덮기
            ctx.fillStyle = '#CBD5E1'; // slate-300
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            ctx.font = "bold 20px Pretendard";
            ctx.fillStyle = "#94A3B8";
            ctx.textAlign = "center";
            ctx.fillText("이곳을 문질러보세요", canvas.width/2, canvas.height/2);

            let isDrawing = false;
            ctx.lineJoin = "round";
            ctx.lineCap = "round";
            ctx.lineWidth = 40; // 지워지는 펜 굵기

            function scratch(e) {
                if (!isDrawing) return;
                e.preventDefault();
                const rect = canvas.getBoundingClientRect();
                const x = (e.clientX || e.touches[0].clientX) - rect.left;
                const y = (e.clientY || e.touches[0].clientY) - rect.top;
                
                ctx.globalCompositeOperation = "destination-out"; // 이 모드가 '지우개' 역할
                ctx.beginPath();
                ctx.arc(x, y, 20, 0, Math.PI * 2, false);
                ctx.fill();

                // 텍스트 보이게 하기
                document.getElementById('manna-ref').style.opacity = '1';
            }

            canvas.addEventListener('mousedown', () => isDrawing = true);
            canvas.addEventListener('mousemove', scratch);
            canvas.addEventListener('mouseup', () => isDrawing = false);
            canvas.addEventListener('touchstart', () => isDrawing = true, {passive: false});
            canvas.addEventListener('touchmove', scratch, {passive: false});
            canvas.addEventListener('touchend', () => isDrawing = false);
        }
